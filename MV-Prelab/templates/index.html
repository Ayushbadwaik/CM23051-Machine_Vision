Source Code Reference
Image Loading & Properties (OpenCV)
# 1. Image Loading import cv2, numpy as np img = cv2.imread("image.jpg") # img is a NumPy array (H x W x 3) in BGR order # 2. Image Properties h, w, c = img.shape # height, width, channels dtype = img.dtype # uint8 nbytes = img.nbytes # total bytes in memory # 3. Displaying with OpenCV cv2.imshow("Window", img) cv2.waitKey(0) cv2.destroyAllWindows()
Color Channel Splitting
# Split into B, G, R channels b, g, r = cv2.split(img) # Reconstruct individual channel views zero = np.zeros_like(b) r_vis = cv2.merge([zero, zero, r]) g_vis = cv2.merge([zero, g, zero]) b_vis = cv2.merge([b, zero, zero]) # Convert to greyscale gray = cv2.cvtColor(img, cv2.COLOR_BGR2GRAY) # Convert to HSV hsv = cv2.cvtColor(img, cv2.COLOR_BGR2HSV) # Channel mean values mean = cv2.mean(img) # (B, G, R, 0)
Webcam / Video Capture
# Access webcam (index 0 = default camera) cap = cv2.VideoCapture(0) while cap.isOpened(): ret, frame = cap.read() if not ret: break # Process frame gray = cv2.cvtColor(frame, cv2.COLOR_BGR2GRAY) cv2.imshow("Webcam", frame) if cv2.waitKey(1) & 0xFF == ord('q'): break cap.release() cv2.destroyAllWindows()
Video File Capture & Saving
# Read from a video file cap = cv2.VideoCapture("video.mp4") fps = cap.get(cv2.CAP_PROP_FPS) w = int(cap.get(cv2.CAP_PROP_FRAME_WIDTH)) h = int(cap.get(cv2.CAP_PROP_FRAME_HEIGHT)) # VideoWriter to save output fourcc = cv2.VideoWriter_fourcc(*'mp4v') out = cv2.VideoWriter('out.mp4', fourcc, fps, (w, h)) while cap.isOpened(): ret, frame = cap.read() if not ret: break out.write(frame) cap.release(); out.release()
